\subsection{Extended Rytov theory}
The received irradiance of an optical wave can be modeled as a modulation process in which small-scale (diffractive) fluctuations are multiplicatively modulated by statistically independent large-scale (refractive) fluctuations:
\begin{align*}
n_{eff} (\textbf{R}) &=  n_X(\textbf{R})+n_Y(\textbf{R})
\end{align*}
$X$ from large scale effects; $Y$ from small scale effects.
\begin{align*}
U(\textbf{r}, L) &= U_0(\textbf{r}, L)\cdot e^{\psi_X(\textbf{r}, L)}\cdot e^{\psi_Y(\textbf{r}, L)}
\end{align*}
$\psi_X$,$\psi_Y$ complex phase perturbations on the propagating wave.\\
\\

\begin{minipage}[b]{0.5\textwidth}
\begin{itemize}
\item $l_o$ inner scale of turbulence;
\item $L_o$ outer scale of turbulence;
\item $\sqrt{L/k}$ first Fresnel zone;
\item $\rho_0$ spatial coherence radius of the optical wave;
\item $L/k\rho_0$ scattering disk;
\end{itemize}
\end{minipage}
\begin{minipage}[c]{0.5\textwidth}
Def. scattering disk: refractive cell size $l$ at which the focusing angle $\theta_F\sim l/L$ is equal to the average scattering angle $\theta_D$.\\

Three regions defined:
\begin{itemize}
\item $l_1\sim\rho_0$; Coherence radius 
\item $l_2\sim\sqrt{L/k}$; Fresnel zone
\item $l_3\sim L/k\rho_0$; Scattering disk
\end{itemize}
\end{minipage}
\\

\subsection{Scintillation index model}
(...) One random process modulating another: (...) the atmosphere is the modulation process of the instantaneous light that passes through it:
\begin{align*}
\hat{I} &= \frac{I}{\langle I \rangle} = XY
\end{align*}
\begin{itemize}
\item $X$ from large scale effects;
\item $Y$ from small scale effects;
\end{itemize}
\textbf{Second moment}
\begin{align*}
\langle \hat{I}^2 \rangle &= \langle X^2 \rangle\langle Y^2 \rangle\\
	&= (1+\sigma_X^2)(1+\sigma_Y^2)
\end{align*}
assuming $\langle X \rangle=\langle Y \rangle=1$ and $\sigma_x^2$, $\sigma_Y^2$ normalized variances
\begin{align*}
\sigma_I^2 &= (1+\sigma_X^2)(1+\sigma_Y^2)-1\\
&= \sigma_X^2+\sigma_Y^2+\sigma_X^2\sigma_Y^2
\end{align*}
using Rytov log amplitude of an optical wave (variance of log irradiance) results:
\begin{align*}
\sigma_I^2 &= \exp(\sigma^2_{\ln X}+\sigma^2_{\ln Y})-1
\end{align*}
then in weak irradiance fluctuations results:
\begin{align*}
\sigma_X^2 &\cong \sigma^2_{\ln X}\\
\sigma_Y^2 &\cong \sigma^2_{\ln Y}\\
\sigma_I^2 &= \sigma^2_{\ln I} = \sigma^2_{\ln X}+\sigma^2_{\ln Y}
\end{align*}

\subsection{Fried parameter (Coherence width)}
For $C_n^2$ constant, horizontal propagation with uniform turbulence:
\begin{align*}
r_0 &= (0.16\cdot C_n^2 k^2 L)^{-3/5}
\end{align*}
\\
For Horizontal link (plane wave):
\begin{align*}
r_0 &= \left(1.46C_n^2k^2L\right)^{-3/5}
\end{align*}
\\
For Vertical link or slant paths:\\
\begin{align*}
r_0 &= \left[0.42\sec(\zeta)k^2\int_{h_0}^H C_n^2(h)dh\right]^{-3/5}
\end{align*}

with $\zeta$ the zenith angle ($\sec(0)=1$), $h_0$ the height above ground level of the uplink transitter and/or downlink receiver, $H=h_0+L\cos(\zeta)$ is the satellite altitude. \\
\\
Downlink can be approximated by a plane wave. Use vertical link formula.\\
Uplink can be approximated by a spherical wave. Use next formula:\\

\begin{align*}
r_0 &= \left[0.42\sec(\zeta)k^2\int_{h_0}^H C_n^2(h)\left(1-\frac{h-h_{OGS}}{H-h_{OGS}}\right)
dh\right]^{-3/5}
\end{align*}
with H the satellite altitude, $h_{OGS}$ the altitude of the OGS receiver, and h the integrand.
